cmake_minimum_required(VERSION 3.14)
project(fpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(fpp SHARED src/world.cpp src/path.cpp src/point_astar.cpp src/theta_star.cpp src/se2_lattice.cpp src/rrt_star.cpp src/c_api.cpp)
target_include_directories(fpp PUBLIC include)
set_target_properties(fpp PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(fpp PRIVATE -Wall -Wextra)

include(CTest)
if(BUILD_TESTING)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.14.0)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(fpp_tests tests/test_fpp.cpp)
  target_link_libraries(fpp_tests PRIVATE fpp GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(fpp_tests)
endif()
